cmake_minimum_required(VERSION 3.20)
project(avtools_native LANGUAGES CXX)
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
  message(FATAL_ERROR "The first validated avTools native target is macOS arm64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
if(NOT EXISTS "${NODE_INCLUDE_DIR}/node_api.h" OR NOT EXISTS "${TEXT_ENGINE_LIBRARY}")
  message(FATAL_ERROR "Set NODE_INCLUDE_DIR and TEXT_ENGINE_LIBRARY to the pinned Node headers and Rust static library")
endif()
add_library(avtools_text MODULE text_addon.cpp)
target_include_directories(avtools_text PRIVATE "${NODE_INCLUDE_DIR}")
target_compile_definitions(avtools_text PRIVATE NAPI_VERSION=8)
target_link_libraries(avtools_text PRIVATE "${TEXT_ENGINE_LIBRARY}" "-framework CoreFoundation" "-framework Security" iconv)
target_link_options(avtools_text PRIVATE "-undefined" "dynamic_lookup")
set_target_properties(avtools_text PROPERTIES PREFIX "" SUFFIX ".node")
